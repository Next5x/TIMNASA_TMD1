```bash

git clone [https://github.com/Next5x/TIMNASA_TMD1.git](https://github.com/Next5x/TIMNASA_TMD1.git)

# 2. Ingia kwenye folder
cd TIMNASA_TMD1

# 3. Weka dependencies
npm install

# 4. Anzisha Bot
npm start
